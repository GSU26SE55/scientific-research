% ---- Encoding --------------------------------------------------------------
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}

% ---- Math ------------------------------------------------------------------
\usepackage{amsmath,amssymb,amsfonts}

% ---- Graphics & figures ----------------------------------------------------
\usepackage{graphicx}
\graphicspath{{figures/}}       % default figure search path
% NOTE: do NOT load `caption'/`subcaption' -- they override the LNCS caption
% layout and emit an "Unknown document class" warning with llncs.cls.

% ---- Float packing ---------------------------------------------------------
% This paper is float-heavy (6 tables + 2 figures). LaTeX's default placement
% heuristics leave large slack (they cap floats at 70 % of a page and reserve
% 20 % for text), which inflates the page count. The parameters below let
% floats pack more densely. They only affect *placement*, not the page geometry
% or type area defined by llncs.cls, so the Springer layout is preserved.
\renewcommand{\topfraction}{0.9}       % up to 90 % of a page top may be floats
\renewcommand{\bottomfraction}{0.8}
\renewcommand{\textfraction}{0.07}     % require only 7 % text on a mixed page
\renewcommand{\floatpagefraction}{0.75}% a float-only page must be >=75 % full
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}
% Trim the (generous) default space between a float and the body text. This is
% a modest, uniform reduction to recover the whitespace LaTeX leaves around the
% paper's 8 floats; it does not alter the type area or line spacing.
\setlength{\textfloatsep}{9pt plus 2pt minus 2pt}
\setlength{\intextsep}{8pt plus 2pt minus 2pt}
\setlength{\floatsep}{8pt plus 2pt minus 2pt}

% ---- Tables ----------------------------------------------------------------
\usepackage{array}
\usepackage{booktabs}           % \toprule, \midrule, \bottomrule
\usepackage{multirow}
\usepackage{tabularx}
\usepackage{ragged2e}           % \RaggedRight inside p/X columns

% New column types used by the wide comparison tables.
\newcolumntype{L}[1]{>{\RaggedRight\arraybackslash}p{#1}}
\newcolumntype{C}[1]{>{\Centering\arraybackslash}p{#1}}
\newcolumntype{Y}{>{\RaggedRight\arraybackslash}X}

% Layout helpers for the wide comparison tables. The LNCS text block is only
% 12.2 cm wide, so multi-column tables need reduced inter-column padding and
% permissive hyphenation inside p/X columns to avoid overfull boxes.
%   \tabtight  -- narrow padding, aggressive line breaking (wide tables)
%   \tabairy   -- extra padding (narrow tables, keeps headers from touching)
\newcommand{\tabtight}{%
  \setlength{\tabcolsep}{3pt}%
  \hyphenpenalty=10 \exhyphenpenalty=10 \tolerance=9999
  \emergencystretch=1em\relax}
\newcommand{\tabairy}{\setlength{\tabcolsep}{9pt}}

% Two-line (stacked) table header, e.g. \hd{MAE}{(\%)} -- keeps wide numeric
% headers from dictating the column width.
\newcommand{\hd}[2]{\begin{tabular}[b]{@{}c@{}}\textbf{#1}\\\textbf{#2}\end{tabular}}

% ---- Line breaking ---------------------------------------------------------
% The LNCS measure is narrow (12.2 cm), so a few justified lines cannot be
% broken without overflowing the margin. \emergencystretch lets TeX loosen
% interword spacing on those lines instead of producing overfull boxes.
\setlength{\emergencystretch}{1.5em}

% Left-aligned explanatory note underneath a table (e.g. symbol footnotes).
% The minipage keeps the text flush-left even inside a centred table float.

  \noindent\begin{minipage}{\linewidth}%
    \setlength{\parindent}{0pt}\small\raggedright #1%
  \end{minipage}\par}
